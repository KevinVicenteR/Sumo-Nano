// Lee el control remoto con filtro de ruido
bool Robot::leerRemoto() {
    const int nivel = digitalRead(Pin_Control_Remoto);
    const bool crudo = REMOTE_ACTIVE_HIGH ? (nivel == HIGH) : (nivel == LOW);
    const unsigned long ahora = millis();
    if (crudo == remotoEstable) {
        remotoCambioDesde = ahora;
    } else if (ahora - remotoCambioDesde >= REMOTO_FILTRO_MS) {
        remotoEstable = crudo;
        remotoCambioDesde = ahora;
    }
    return remotoEstable;
}

bool Robot::pisoIncierto(int valor, int negro, bool valido) const {
    const int diferencia = PISO_BLANCO_LOW ? valor - negro : negro - valor;
    return valido && diferencia > PISO_NEGRO_TOLERANCIA_ADC;
}

bool Robot::leerPiso(bool &pisoIzq, bool &pisoDer) {
    const int valorIzq = sensorPisoIzq.leerValor();
    const int valorDer = sensorPisoDer.leerValor();
    pisoIzq = sensorPisoIzq.esBlanco(valorIzq) || pisoIncierto(valorIzq, negroIzq, negroIzqValido);
    pisoDer = sensorPisoDer.esBlanco(valorDer) || pisoIncierto(valorDer, negroDer, negroDerValido);
    return pisoIzq || pisoDer;
}

void Robot::predecirBorde(int valorIzq, int valorDer, unsigned long ahora,
                         bool &riesgoIzq, bool &riesgoDer) {
    riesgoIzq = riesgoDer = false;
    const unsigned long dt = ahora - ultimaMuestraPiso;
    if (pisoPrevioValido && dt < PREDICCION_MUESTREO_MS) return;
    if (pisoPrevioValido && dt <= 20 && PREDICCION_BORDE_MS > 0) {
        const int valores[2] = {valorIzq, valorDer};
        const int previos[2] = {pisoPrevioIzq, pisoPrevioDer};
        bool *riesgos[2] = {&riesgoIzq, &riesgoDer};
        for (uint8_t i = 0; i < 2; ++i) {
            const int distanciaADC = PISO_BLANCO_LOW ? valores[i] - BLANCO : BLANCO - valores[i];
            const int acercamiento = PISO_BLANCO_LOW ? previos[i] - valores[i] : valores[i] - previos[i];
            // Sin coordenadas ni reloj del simulador: extrapolar solo lecturas reales.
            *riesgos[i] = distanciaADC > 0 && distanciaADC <= PREDICCION_MARGEN_ADC &&
                acercamiento >= PREDICCION_CAMBIO_MIN_ADC &&
                (long)distanciaADC * (long)dt <= (long)acercamiento * PREDICCION_BORDE_MS;
        }
    }
    pisoPrevioIzq = valorIzq;
    pisoPrevioDer = valorDer;
    ultimaMuestraPiso = ahora;
    pisoPrevioValido = true;
    // Aprender el negro lejos de la transición, sin incorporar lecturas inciertas.
    const int valores[2] = {valorIzq, valorDer};
    int *negros[2] = {&negroIzq, &negroDer};
    bool *validos[2] = {&negroIzqValido, &negroDerValido};
    for (uint8_t i = 0; i < 2; ++i) {
        const int distanciaADC = PISO_BLANCO_LOW ? valores[i] - BLANCO : BLANCO - valores[i];
        if (distanciaADC > PREDICCION_MARGEN_ADC &&
            !pisoIncierto(valores[i], *negros[i], *validos[i])) {
            *negros[i] = *validos[i] ? ((long)*negros[i] * 7 + valores[i]) / 8 : valores[i];
            *validos[i] = true;
        }
    }
}

void Robot::reiniciarEstado() {
    avanceSuaveValido = false;
    centralReciente = false;
    pisoPrevioValido = false;
    negroIzqValido = negroDerValido = false;
    bordeDetectado = false;
    busquedaDerecha = true;
    faseBusqueda = 0;
    inicioFase = 0;
    huboContacto = false;
    ultimoContacto = 0;
    ultimoContactoFrontal = false;
    ultimoCambioGiro = 0;
    seguimientoActivo = false;
    desatascoActivo = false;
    ataqueActivo = false;
    lateralActivo = false;
}

bool Robot::enemigoDetectado() {
    return sensorFrontal.detectar() || sensorFrontalDer.detectar() ||
           sensorFrontalIzq.detectar() || sensorLateralDer.detectar() ||
           sensorLateralIzq.detectar();
}

// Retrocede hasta retirar los sensores del blanco, con límite de tiempo.
bool Robot::retrocesoSeguro(unsigned long duracionMs) {
    return retrocesoSeguro(duracionMs, 0);
}

bool Robot::retrocesoSeguro(unsigned long duracionMs, unsigned long retiradaMinMs) {
    const unsigned long inicio = millis();
    unsigned long libreDesde = inicio;
    bool libre = false;
    while (millis() - inicio < duracionMs) {
        if (!leerRemoto()) {
            motores.detener();
            return false;
        }
        retroceder();
        bool pisoIzq, pisoDer;
        const unsigned long ahora = millis();
        if (leerPiso(pisoIzq, pisoDer)) {
            libre = false;
        } else if (!libre) {
            libre = true;
            libreDesde = ahora;
        }
        if (libre && ahora - libreDesde >= PISO_LIBRE_MS) {
            // Con piso estable, ceder al siguiente loop para perseguir al rival.
            if (ahora - inicio >= retiradaMinMs && enemigoDetectado()) {
                motores.detener();
                return false;
            }
            if (ahora - inicio >= RETROCESO_MIN_MS) return true;
        }
    }
    return true;
}

// Permite que caiga la velocidad residual antes de cambiar de maniobra.
// STOP y enemigo con piso libre pueden cancelar el resto del escape.
bool Robot::pausaEscape(unsigned long duracionMs) {
    return pausaEscape(duracionMs, true);
}

bool Robot::pausaEscape(unsigned long duracionMs, bool permitirEnemigo) {
    motores.detener();
    ESTADO_SIM(8);
    const unsigned long inicio = millis();
    while (millis() - inicio < duracionMs) {
        if (!leerRemoto()) return false;
        bool izquierda, derecha;
        if (!leerPiso(izquierda, derecha) && permitirEnemigo && enemigoDetectado()) return false;
    }
    return true;
}

// Gira sobre su eje para alejarse del borde
bool Robot::giroEscape(bool haciaDerecha, unsigned long duracionMs) {
    const unsigned long inicio = millis();
    while (millis() - inicio < duracionMs) {
        if (!leerRemoto()) {
            motores.detener();
            return false;
        }
        bool pisoIzq = false;
        bool pisoDer = false;
        if (leerPiso(pisoIzq, pisoDer)) {
            return true;
        }
        if (enemigoDetectado()) {
            motores.detener();
            return false;
        }
        ESTADO_SIM(6);
        if (haciaDerecha) motores.derecha(VELOCIDAD_GIRO_ESCAPE);
        else motores.izquierda(VELOCIDAD_GIRO_ESCAPE);
    }
    return true;
}

void Robot::girarHacia(bool haciaDerecha) {
    if (haciaDerecha) {
        moverDerecha();
    } else {
        moverIzquierda();
    }
}

void Robot::setup() {
    pinMode(S_PISO_IZQ, INPUT);
    pinMode(S_PISO_DER, INPUT);
    pinMode(S_FRONT_DER, INPUT);
    pinMode(S_FRONT_CEN, INPUT);
    pinMode(S_FRONT_IZQ, INPUT);
    pinMode(S_LAT_IZQ, INPUT);
    pinMode(S_LAT_DER, INPUT);
    pinMode(Pin_Control_Remoto, INPUT);
    motores.iniciar();
#if MODO_CALIBRACION
    Serial.begin(9600);
#endif
}

void Robot::detenerse() {
    avanceSuaveValido = false;
    motores.detener();
}

void Robot::avanzarFluido(int izquierda, int derecha) {
    const unsigned long ahora = millis();
    if (!avanceSuaveValido) {
        // Retomar la orden directamente tras una maniobra de seguridad.
        avancePWM_Izq = izquierda;
        avancePWM_Der = derecha;
        avanceSuaveValido = true;
        ultimaRampa = ahora;
    } else if (ahora != ultimaRampa) {
        const unsigned long dt = ahora - ultimaRampa;
        const int paso = dt > 255UL / RAMPA_AVANCE_PWM_MS ? 255 : dt * RAMPA_AVANCE_PWM_MS;
        avancePWM_Izq += constrain(izquierda - avancePWM_Izq, -paso, paso);
        avancePWM_Der += constrain(derecha - avancePWM_Der, -paso, paso);
        ultimaRampa = ahora;
    }
    motores.diferencial(avancePWM_Izq, avancePWM_Der);
}

void Robot::ataqueEnemigo() {
    // Ataque continuo. Blanco y STOP se revisan en el siguiente ciclo,
    // sin esperas ni frenadas periódicas entre detecciones centrales.
    ESTADO_SIM(4);
    avanzarFluido(Velocidad_maxima_Ataque, Velocidad_maxima_Ataque);
}

void Robot::moverAdelante() {
    avanzarFluido(Velocidad_movimiento_seguir, Velocidad_movimiento_seguir);
}

void Robot::retroceder() {
    avanceSuaveValido = false;
    ESTADO_SIM(5);
    motores.retroceder(Velocidad_estandar);
}

void Robot::moverDerecha() {
    avanceSuaveValido = false;
    motores.derecha(Velocidad_maxima);
}

void Robot::moverIzquierda() {
    avanceSuaveValido = false;
    motores.izquierda(Velocidad_maxima);
}

void Robot::sensoresPiso(bool pisoIzq, bool pisoDer) {
    sensoresPiso(pisoIzq, pisoDer, false);
}

void Robot::sensoresPiso(bool pisoIzq, bool pisoDer, bool anticipado) {
    avanceSuaveValido = false;
    centralReciente = false;
    static bool giroAlternadoDerecha = true;

    // Detener el giro residual antes de retroceder evita escapar en un arco
    // hacia el exterior cuando se detecta blanco durante una orientación.
    // Una predicción todavía no es blanco: completar el freno y una retirada
    // breve evita que el enemigo cancele la prevención en la misma lectura.
    if (!pausaEscape(PAUSA_ESCAPE_MS, !anticipado)) return;
    if (!retrocesoSeguro(TIEMPO_RETROCESO_MS, anticipado ? PREDICCION_RETIRADA_MIN_MS : 0)) return;
    if (!pausaEscape(PAUSA_RETROCESO_MS)) return;

    if (pisoIzq && pisoDer) {
        const bool completado = giroEscape(giroAlternadoDerecha, TIEMPO_GIRO_BORDE_FRENTE_MS);
        giroAlternadoDerecha = !giroAlternadoDerecha;
        if (!completado) return;
    } else if (pisoDer) {
        if (!giroEscape(false, TIEMPO_GIRO_BORDE_LADO_MS)) return;
    } else if (pisoIzq) {
        if (!giroEscape(true, TIEMPO_GIRO_BORDE_LADO_MS)) return;
    }
    pausaEscape(PAUSA_GIRO_MS);
}

// Mantiene una decisión breve para evitar alternar izquierda/derecha con ruido.
// Reposiciona solo lecturas laterales o ambiguas persistentes; un frontal
// inequívoco mantiene la persecución y no se interrumpe por un temporizador.
void Robot::seguirDireccion(bool haciaDerecha, bool ambiguo, bool frontal) {
    const unsigned long ahora = millis();
    if (!seguimientoActivo) {
        seguimientoActivo = true;
        inicioSeguimiento = ahora;
        if (ultimoCambioGiro == 0 || ahora - ultimoCambioGiro >= GIRO_COMPROMISO_MS) {
            ultimoCambioGiro = ahora;
            giroSeguimientoDerecha = ambiguo ? busquedaDerecha : haciaDerecha;
        }
    } else if (!ambiguo && haciaDerecha != giroSeguimientoDerecha &&
               ahora - ultimoCambioGiro >= GIRO_COMPROMISO_MS) {
        giroSeguimientoDerecha = haciaDerecha;
        ultimoCambioGiro = ahora;
    }
    if ((!frontal || ambiguo) && ahora - inicioSeguimiento >= GIRO_MAX_SEGUIMIENTO_MS) {
        desatascoActivo = true;
        inicioDesatasco = ahora;
        seguimientoActivo = false;
        ESTADO_SIM(7);
        avanzarFluido(VELOCIDAD_DESATASCO, VELOCIDAD_DESATASCO);
        return;
    }
    ESTADO_SIM(3);
    if (frontal) {
        if (giroSeguimientoDerecha) avanzarFluido(Velocidad_maxima, VELOCIDAD_CURVA_INTERIOR);
        else avanzarFluido(VELOCIDAD_CURVA_INTERIOR, Velocidad_maxima);
    } else {
        girarHacia(giroSeguimientoDerecha);
    }
}

void Robot::sensoresFrontales(bool derecho, bool izquierdo) {
    seguirDireccion(derecho, derecho && izquierdo, true);
}

void Robot::sensoresLaterales(bool izquierdo, bool derecho) {
    const unsigned long ahora = millis();
    if (!lateralActivo) {
        if (izquierdo == derecho) {
            // Ambos lados son ambiguos: conservar la salida de desbloqueo.
            seguirDireccion(derecho, true, false);
            return;
        }
        lateralActivo = true;
        avanceSuaveValido = false;
        lateralDerecha = derecho;
        inicioLateral = ahora;
        seguimientoActivo = false;
        ultimoCambioGiro = 0;
        busquedaDerecha = derecho;
    }
    const unsigned long transcurrido = ahora - inicioLateral;
    if (transcurrido < FRENO_LATERAL_MS) {
        motores.detener();
        ESTADO_SIM(10);
        return;
    }
    if (transcurrido - FRENO_LATERAL_MS < GIRO_LATERAL_MAX_MS) {
        if (lateralDerecha) motores.derecha(VELOCIDAD_GIRO_LATERAL);
        else motores.izquierda(VELOCIDAD_GIRO_LATERAL);
        ESTADO_SIM(11);
        return;
    }
    // Una señal lateral persistente no puede mantener un giro infinito.
    lateralActivo = false;
    desatascoActivo = true;
    inicioDesatasco = ahora;
    motores.adelante(VELOCIDAD_DESATASCO);
    ESTADO_SIM(7);
}

#if MODO_CALIBRACION
// Muestra las lecturas de los sensores por Serial
void Robot::calibrar() {
    static unsigned long ultimoEnvio = 0;
    motores.detener();
    if (millis() - ultimoEnvio < 100) return;
    ultimoEnvio = millis();

    Serial.print(F("Piso I:"));
    Serial.print(sensorPisoIzq.leerValor());
    Serial.print(F(" D:"));
    Serial.print(sensorPisoDer.leerValor());
    Serial.print(F(" | Frontal I:"));
    Serial.print(sensorFrontalIzq.detectar());
    Serial.print(F(" C:"));
    Serial.print(sensorFrontal.detectar());
    Serial.print(F(" D:"));
    Serial.print(sensorFrontalDer.detectar());
    Serial.print(F(" | Lateral I:"));
    Serial.print(sensorLateralIzq.detectar());
    Serial.print(F(" D:"));
    Serial.print(sensorLateralDer.detectar());
    Serial.print(F(" | Remoto:"));
    Serial.println(leerRemoto() ? F("START") : F("STOP"));
}
#endif

void Robot::loop() {
#if MODO_CALIBRACION
    calibrar();
    return;
#endif

    // Control remoto
    if (!leerRemoto()) {
        avanceSuaveValido = false;
        centralReciente = false;
        ESTADO_SIM(0);
        motores.detener();
        estadoAnterior = false;
        return;
    }
    if (!estadoAnterior) {
        reiniciarEstado();
        estadoAnterior = true;
    }

    // Lectura de sensores
    const int valorPisoIzq = sensorPisoIzq.leerValor();
    const int valorPisoDer = sensorPisoDer.leerValor();
    const bool PisoIzq = sensorPisoIzq.esBlanco(valorPisoIzq) || pisoIncierto(valorPisoIzq, negroIzq, negroIzqValido);
    const bool PisoDer = sensorPisoDer.esBlanco(valorPisoDer) || pisoIncierto(valorPisoDer, negroDer, negroDerValido);

    const bool FrontalDer = sensorFrontalDer.detectar();
    const bool FrontalIzq = sensorFrontalIzq.detectar();
    const bool FrontalCentral = sensorFrontal.detectar();
    const bool LateralDer = sensorLateralDer.detectar();
    const bool LateralIzq = sensorLateralIzq.detectar();

    const unsigned long ahora = millis();
    bool riesgoIzq, riesgoDer;
    predecirBorde(valorPisoIzq, valorPisoDer, ahora, riesgoIzq, riesgoDer);
    if (FrontalDer || FrontalIzq || FrontalCentral || LateralDer || LateralIzq) {
        huboContacto = true;
        ultimoContacto = ahora;
        ultimoContactoFrontal = FrontalDer || FrontalIzq || FrontalCentral;
        // Último lado donde se vio al enemigo
        const bool vistoDer = FrontalDer || LateralDer;
        const bool vistoIzq = FrontalIzq || LateralIzq;
        if (vistoDer != vistoIzq) {
            busquedaDerecha = vistoDer;
        }
    }

    // Escape del borde
    if (PisoIzq || PisoDer || riesgoIzq || riesgoDer) {
        ataqueActivo = false;
        bordeDetectado = true;
        seguimientoActivo = false;
        desatascoActivo = false;
        lateralActivo = false;
        sensoresPiso(PisoIzq || riesgoIzq, PisoDer || riesgoDer, !PisoIzq && !PisoDer);
        pisoPrevioValido = false;
        return;
    }

    // El frontal interrumpe inmediatamente el frenado o giro lateral.
    if (FrontalCentral || FrontalDer || FrontalIzq) {
        if (lateralActivo) {
            seguimientoActivo = false;
            ultimoCambioGiro = 0;
        }
        lateralActivo = false;
    }

    if (FrontalCentral) {
        centralReciente = true;
        ultimoCentral = ahora;
    }
    // Conservar el avance ante pérdidas breves del central, salvo un rival lateral.
    const bool mantenerAtaque = centralReciente && ahora - ultimoCentral < MEMORIA_ATAQUE_MS &&
                                !LateralDer && !LateralIzq;
    if (FrontalCentral || mantenerAtaque || (ATAQUE_DOBLE_FRONTAL && FrontalDer && FrontalIzq && !LateralDer && !LateralIzq)) {
        desatascoActivo = false;
        if (FrontalDer != FrontalIzq) {
            // El central confirma ataque, pero el frontal lateral permite
            // centrar al rival sin detenerse ni esperar a perderlo de vista.
            seguirDireccion(FrontalDer, false, true);
            ESTADO_SIM(4);
            if (giroSeguimientoDerecha) avanzarFluido(Velocidad_maxima_Ataque, VELOCIDAD_ATAQUE_CURVA_INTERIOR);
            else avanzarFluido(VELOCIDAD_ATAQUE_CURVA_INTERIOR, Velocidad_maxima_Ataque);
        } else {
            seguimientoActivo = false;
            ataqueEnemigo();
        }
        return;
    }
    ataqueActivo = false;

    if (FrontalDer != FrontalIzq) desatascoActivo = false;

    // Avance breve de desbloqueo. Se relee el borde en cada loop y el central
    // puede interrumpirlo para atacar; no añade otra espera bloqueante.
    if (desatascoActivo) {
        if (ahora - inicioDesatasco < DESATASCO_AVANCE_MS) {
            ESTADO_SIM(7);
            avanzarFluido(VELOCIDAD_DESATASCO, VELOCIDAD_DESATASCO);
            return;
        }
        desatascoActivo = false;
    }

    // Enemigo en frontal izquierdo o derecho
    if (FrontalDer || FrontalIzq) {
        sensoresFrontales(FrontalDer, FrontalIzq);
        return;
    }

    // Atender los laterales desde el arranque, antes de cualquier búsqueda.
    if (lateralActivo || LateralDer || LateralIzq) {
        sensoresLaterales(LateralIzq, LateralDer);
        return;
    }

    // Búsqueda
    const bool cercaBordeIzq = PISO_BLANCO_LOW
        ? valorPisoIzq <= (BLANCO + MARGEN_CERCA_BORDE)
        : valorPisoIzq >= (BLANCO - MARGEN_CERCA_BORDE);
    const bool cercaBordeDer = PISO_BLANCO_LOW
        ? valorPisoDer <= (BLANCO + MARGEN_CERCA_BORDE)
        : valorPisoDer >= (BLANCO - MARGEN_CERCA_BORDE);

    // Se aleja del borde cercano
    if (cercaBordeIzq || cercaBordeDer) {
        seguimientoActivo = false;
        ESTADO_SIM(1);
        if (cercaBordeIzq && !cercaBordeDer) {
            moverDerecha();
        } else if (cercaBordeDer && !cercaBordeIzq) {
            moverIzquierda();
        } else {
            girarHacia(busquedaDerecha);
        }
        faseBusqueda = 0;
        inicioFase = ahora;
        return;
    }

    // Gira hacia el último lado donde se vio al enemigo
    if (huboContacto && (ahora - ultimoContacto) < TIEMPO_REBUSQUEDA_MS) {
        seguirDireccion(busquedaDerecha, false, ultimoContactoFrontal);
        faseBusqueda = 0;
        inicioFase = ahora;
        return;
    }

    seguimientoActivo = false;

    // Patrón de búsqueda: giro, avance, giro, avance
    const unsigned long duraciones[4] = {
        TIEMPO_BUSQUEDA_GIRO_MS, TIEMPO_BUSQUEDA_AVANCE_MS,
        TIEMPO_BUSQUEDA_GIRO_MS, TIEMPO_BUSQUEDA_AVANCE_MS
    };

    if (inicioFase == 0) {
        inicioFase = ahora;
    }
    if ((ahora - inicioFase) >= duraciones[faseBusqueda]) {
        inicioFase = ahora;
        faseBusqueda = (faseBusqueda + 1) % 4;
    }

    if (faseBusqueda == 0 || faseBusqueda == 2) {
        ESTADO_SIM(1);
#if PATRON_BUSQUEDA == 1
        // Barridos a ambos lados con desplazamiento entre ellos.
#if BUSQUEDA_FLUIDA
        const bool derecha = faseBusqueda == 0 ? busquedaDerecha : !busquedaDerecha;
        if (derecha) avanzarFluido(VELOCIDAD_BUSQUEDA, VELOCIDAD_CURVA_INTERIOR);
        else avanzarFluido(VELOCIDAD_CURVA_INTERIOR, VELOCIDAD_BUSQUEDA);
#else
        girarHacia(faseBusqueda == 0 ? busquedaDerecha : !busquedaDerecha);
#endif
#elif PATRON_BUSQUEDA == 2
        if (busquedaDerecha) motores.curvaDerecha(Velocidad_maxima);
        else motores.curvaIzquierda(Velocidad_maxima);
#else
        girarHacia(busquedaDerecha);
#endif
    } else {
        ESTADO_SIM(2);
        avanzarFluido(VELOCIDAD_BUSQUEDA, VELOCIDAD_BUSQUEDA);
    }
}
